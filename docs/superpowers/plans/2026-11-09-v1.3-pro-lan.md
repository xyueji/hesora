```ts
export type LicenseFeature = 'lan_collaboration' | 'cloud_backup' | 'multi_warehouse' | 'rbac' | 'device_handoff'
export type LicenseEdition = 'standard' | 'professional'

export interface LicensePayload {
  customer: string
  issuedAt: string
  expiresAt?: string | null
  machine?: string | null
  edition?: LicenseEdition
  features?: LicenseFeature[]
  maxDevices?: number
}

/** 旧授权码缺省字段 → standard 空特征。所有判定一律走此函数，不直接读 payload。 */
export function resolveLicenseFeatures(p: LicensePayload | null | undefined): {
  edition: LicenseEdition
  features: LicenseFeature[]
  maxDevices: number
} {
  if (!p || p.edition === undefined) return { edition: 'standard', features: [], maxDevices: 1 }
  return {
    edition: p.edition,
    features: Array.isArray(p.features) ? p.features : [],
    maxDevices: typeof p.maxDevices === 'number' && p.maxDevices >= 1 ? p.maxDevices : 1,
  }
}

export function hasFeature(p: LicensePayload | null | undefined, f: LicenseFeature): boolean {
  return resolveLicenseFeatures(p).features.includes(f)
}
```

```ts
import type { Database as BetterDb } from 'better-sqlite3'
import { createHash } from 'node:crypto'
import type { Result } from '../../shared/result'

export function requestHashOf(channel: string, payload: unknown): string {
  return createHash('sha256').update(channel + '\n' + JSON.stringify(payload ?? null)).digest('hex')
}

export interface CommandHit {
  result: Result<unknown>
  replayed: true
  conflict?: false
} | { result: null; replayed: false; conflict: true }

export function commandResultLookup(
  db: BetterDb, requestId: string, channel: string, requestHash?: string
): { result: Result<unknown>; replayed: boolean; conflict: boolean } | null {
  const row = db.prepare('SELECT result_json, request_hash FROM command_results WHERE request_id=? AND channel=?')
    .get(requestId, channel) as { result_json: string; request_hash: string } | undefined
  if (!row) return null
  if (requestHash !== undefined && requestHash !== row.request_hash) {
    return { result: null, replayed: false, conflict: true }
  }
  return { result: JSON.parse(row.result_json) as Result<unknown>, replayed: true, conflict: false }
}

export function recordCommandResult(
  db: BetterDb, requestId: string, channel: string, deviceId: string, result: Result<unknown>, requestHash?: string, now = new Date().toISOString()
): void {
  db.prepare(`INSERT OR IGNORE INTO command_results
    (request_id, channel, device_id, request_hash, result_json, created_at)
    VALUES (?,?,?,?,?,?)`)
    .run(requestId, channel, deviceId, requestHash ?? requestHashOf(channel, undefined), JSON.stringify(result), now)
}

export function purgeCommandResults(db: BetterDb, keepDays = 90, perDeviceCap = 1000): number {
  // 90 天清理 + 每设备保留最近 perDeviceCap 条（created_at 降序）
  const cutoff = new Date(Date.now() - keepDays * 86400000).toISOString()
  db.prepare('DELETE FROM command_results WHERE created_at < ?').run(cutoff)
  const devices = (db.prepare('SELECT DISTINCT device_id FROM command_results').all() as { device_id: string }[])
    .map((r) => r.device_id)
  let removed = 0
  for (const d of devices) {
    removed += db.prepare(`DELETE FROM command_results WHERE device_id=? AND request_id NOT IN (
      SELECT request_id FROM command_results WHERE device_id=? ORDER BY created_at DESC, request_id DESC LIMIT ?)`
    ).run(d, d, perDeviceCap).changes
  }
  return removed
}
```

```ts
export function registerGatewayRoutes(app: FastifyInstance, deps: GatewayDeps, hooks: {
  onChannelSuccess?: (channel: string, payload: unknown, result: Result<unknown>) => void
}): void {
  const wrapped = new Map<string, ReturnType<typeof wrapHandler>>()
  for (const [channel, handler] of Object.entries(deps.handlerMap)) wrapped.set(channel, wrapHandler(channel, handler))

  app.post('/rpc/:channel', async (request, reply) => {
    const channel = (request.params as { channel: string }).channel
    const session = deps.sessionOf(request.headers)
    if (!session) { reply.code(401); return fail('SESSION_EXPIRED', '会话无效或已过期') }
    const handler = wrapped.get(channel)
    if (!handler) { reply.code(404); return fail('E_UNKNOWN_CHANNEL', `通道不存在: ${channel}`) }
    const perm = PERMISSION_MAP[channel]
    if (perm !== undefined && !deps.canFor(session, perm, session.permissions)) {
      return fail('NO_PERMISSION', '没有执行此操作的权限')
    }
    const payload = (request.body as { payload?: unknown } | undefined)?.payload
    const isWrite = IDEMPOTENT_CHANNELS.has(channel)
    const requestId = isWrite ? deps.requestIdOf(request.headers) : undefined
    if (isWrite && !requestId) return fail('E_REQUEST_ID_REQUIRED', '写请求缺少 X-Request-Id')

    if (isWrite && requestId) {
      const hash = requestHashOf(channel, payload)
      const hit = commandResultLookup(deps.getDb(), requestId, channel, hash)
      if (hit?.conflict) return fail('E_REQUEST_ID_CONFLICT', '相同请求ID但内容不同，请更换ID并人工核查')
      if (hit?.replayed) return { result: hit.result, replayed: true }
      // 业务执行 + 防重记录同事务（spec §4.4：better-sqlite3 嵌套事务=savepoint，语义安全）
      const result = deps.getDb().transaction(() => handler(null, payload))() as Result
      if (result.ok || DETERMINISTIC_CODES.has(result.code)) {
        recordCommandResult(deps.getDb(), requestId, channel, session.deviceId, result, hash)
      }
      hooks.onChannelSuccess?.(channel, payload, result)
      return { result }
    }

    const result = await handler(null, payload) as Result
    hooks.onChannelSuccess?.(channel, payload, result)
    return { result }
  })
}

/** 确定性业务失败也登记防重（spec §4.4 步骤 2） */
const DETERMINISTIC_CODES = new Set(['NOT_ENOUGH', 'NO_PERMISSION', 'VALIDATION', 'FORBIDDEN', 'DUP_NAME', 'NOT_FOUND'])
```

注意：`handler` 是 async 时 `db.transaction(() => handler(...))` 不成立（better-sqlite3 事务要求同步函数）。**实现规则：** 写通道 handler 必须同步执行——现有全部写 handler 都是同步 service 调用（better-sqlite3 同步 API），gateway 用 `const result = deps.getDb().transaction(() => wrappedHandlerSync(...))()`；若发现某写 handler 是 async，在本任务中把它改为同步或在该 channel 上禁用幂等并记录 TODO 清单。Task 2 平移时已确认 handler 均同步。
